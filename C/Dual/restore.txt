// Si estamos enviando datos y no han pasado 10 segundos
        if (sending_data) {
            uint32_t current_time = to_ms_since_boot(get_absolute_time());
            
            // Lee el valor del ADC (valor de la señal de AD8231)
            uint16_t adc_value = adc_read(); // El valor del ADC es un número entre 0 y 4095

            // Prepara el valor en formato de cadena para enviarlo
            char buffer[6];  // Espacio suficiente para almacenar el número de 4 dígitos más el salto de línea
            snprintf(buffer, sizeof(buffer), "%u\n", adc_value);

            // Envía el valor por Bluetooth (UART)
            uart_puts(UART_ID, buffer);

            // Verifica si han pasado 10 segundos (10000 ms)
            if (current_time - start_time >= 10000) {
                sending_data = false;  // Detén el envío después de 10 segundos
            }
        }
